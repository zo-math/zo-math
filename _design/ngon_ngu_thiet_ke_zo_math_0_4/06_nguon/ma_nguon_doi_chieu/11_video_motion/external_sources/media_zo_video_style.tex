\newcommand{\zoMakeBox}[6]{%
  \setbox#1=\vbox{\hsize=#2bp\parindent=0pt\raggedright
    \rightskip=0pt plus .5\hsize\parfillskip=0pt\tolerance=10000
    \hyphenpenalty=10000\exhyphenpenalty=10000\emergencystretch=2em
    \fontsize{#4bp}{#5bp}\selectfont\color{zoText}#6\par}%
  \ifdim\dimexpr\ht#1+\dp#1\relax>#3bp
    \PackageError{ZO}{ZO-TEXT-TOO-TALL}{Rut gon chu de hoac cau dan.}%
  \fi
}

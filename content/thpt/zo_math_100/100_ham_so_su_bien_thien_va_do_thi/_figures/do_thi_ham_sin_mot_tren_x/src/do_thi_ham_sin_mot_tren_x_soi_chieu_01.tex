% A finite number of oscillations is shown on each side of the origin.
\pgfmathsetmacro{\zoUMin}{pi}
\pgfmathsetmacro{\zoUMax}{10*pi}

\begin{document}
\begin{tikzpicture}[zo graph field]
  \begin{axis}[
    zo axis,
    xmin=-0.35,
    xmax=0.35,
    ymin=-1.18,
    ymax=1.18,
    xtick={-0.31831,-0.15915,0.15915,0.31831},
    ytick={-1,1},
    xticklabel=\empty,
    yticklabel=\empty,
    xlabel={$x$},
    ylabel={$y$},
    xlabel style={
      at={(axis description cs:1.025,0.5)},
      anchor=west,
      text=zoText,
      font=\normalsize
    },
    ylabel style={
      at={(axis description cs:0.5,1.045)},
      anchor=south,
      text=zoText,
      font=\normalsize,
      rotate=0
    }
  ]
    % Reference envelope for x sin(1/x).
    \addplot[
      zo reference,
      domain=-0.35:0.35,
      samples=2
    ] {x};

    \addplot[
      zo reference,
      domain=-0.35:0.35,
      samples=2
    ] {-x};

    % Secondary graph: g(x)=x sin(1/x), parameterized by u=1/x.
    \addplot[
      zo graph secondary,
      parametric,
      variable=\u,
      domain=\zoUMin:\zoUMax,
      samples=721
    ] ({1/\u},{sin(deg(\u))/\u});

    \addplot[
      zo graph secondary,
      parametric,
      variable=\u,
      domain=\zoUMin:\zoUMax,
      samples=721
    ] ({-1/\u},{sin(deg(\u))/\u});

    % Main graph: f(x)=sin(1/x), split at x=0 and parameterized by u.
    \addplot[
      zo graph main,
      parametric,
      variable=\u,
      domain=\zoUMin:\zoUMax,
      samples=721
    ] ({1/\u},{sin(deg(\u))});

    \addplot[
      zo graph main,
      parametric,
      variable=\u,
      domain=\zoUMin:\zoUMax,
      samples=721
    ] ({-1/\u},{-sin(deg(\u))});

    % Axis labels: every displayed value is an independent node.
    \node[
      text=zoText,
      font=\normalsize,
      anchor=north east,
      xshift=-2pt,
      yshift=-1pt
    ] at (axis cs:0,0) {$O$};

    % SỬA 2:
    % Đổi bốn nhãn mốc trên trục hoành từ
    % \footnotesize sang \small để đồng bộ mẫu tham chiếu.
    \node[
      text=zoText,
      font=\small,
      anchor=north,
      xshift=0pt,
      yshift=-1pt
    ] at (axis cs:{-1/pi},0) {$-\frac{1}{\pi}$};

    \node[
      text=zoText,
      font=\small,
      anchor=north,
      xshift=-8pt,
      yshift=-1pt
    ] at (axis cs:{-1/(2*pi)},0) {$-\frac{1}{2\pi}$};

    \node[
      text=zoText,
      font=\small,
      anchor=south,
      xshift=5pt,
      yshift=1pt
    ] at (axis cs:{1/(2*pi)},0) {$\frac{1}{2\pi}$};

    \node[
      text=zoText,
      font=\small,
      anchor=south,
      xshift=0pt,
      yshift=1pt
    ] at (axis cs:{1/pi},0) {$\frac{1}{\pi}$};

    \node[
      text=zoText,
      font=\small,
      anchor=east,
      xshift=-2pt,
      yshift=0pt
    ] at (axis cs:0,-1.05) {$-1$};

    \node[
      text=zoText,
      font=\small,
      anchor=east,
      xshift=-2pt,
      yshift=0pt
    ] at (axis cs:0,1.05) {$1$};

    \node[
      rotate=12,
      text=zoText,
      font=\small,
      anchor=south west,
      xshift=2pt,
      yshift=1pt
    ] at (axis cs:0.25,0.25) {$y=x$};

    \node[
      rotate=-12,
      text=zoText,
      font=\small,
      anchor=north west,
      xshift=2pt,
      yshift=-1pt
    ] at (axis cs:0.25,-0.25) {$y=-x$};

    % Legend for the two functions under comparison.
    \matrix[
      anchor=north east,
      at={(rel axis cs:1,1)},
      fill=zoWhite,
      draw=zoPlotBorder,
      line width=0.45pt,
      rounded corners=1mm,
      inner sep=5pt,
      row sep=3pt,
      column sep=5pt,
      ampersand replacement=\&,
      nodes={
        anchor=west,
        inner sep=0pt,
        outer sep=0pt
      }
    ] {
      \node[
        draw=zoGraphMain,
        fill=zoGraphMain,
        line width=0.4pt,
        minimum width=5.5pt,
        minimum height=5.5pt
      ] {}; \&
      \node[
        text=zoText,
        font=\small
      ] {$y=\sin\!\left(\dfrac{1}{x}\right)$}; \\

      \node[
        draw=zoGraphStrong,
        fill=zoGraphStrong,
        line width=0.4pt,
        minimum width=5.5pt,
        minimum height=5.5pt
      ] {}; \&
      \node[
        text=zoText,
        font=\small
      ] {$y=x\sin\!\left(\dfrac{1}{x}\right)$}; \\
    };
  \end{axis}
\end{tikzpicture}

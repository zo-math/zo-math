    show background rectangle,

% The tails are shown up to x=-3 and x=3; near 0, only finitely many
% oscillations are displayed on each side of the origin.
\pgfmathsetmacro{\zoUMin}{1/3}
\pgfmathsetmacro{\zoUMax}{10*pi}

\begin{document}
\begin{tikzpicture}[zo graph field]
  \begin{axis}[
    zo axis,
    xmin=-1.2,
    xmax=1.2,
    ymin=-1.18,
    ymax=1.18,
    xtick={-0.31831,-0.15915,0.15915,0.31831},
    ytick={-1,1},
    xticklabel=\empty,
    yticklabel=\empty,
    xlabel={$x$},
    ylabel={$y$},
    xlabel style={
      at={(axis description cs:1.025,0.5)},
      anchor=west,
      text=zoText,
      font=\normalsize
    },
    ylabel style={
      at={(axis description cs:0.5,1.045)},
      anchor=south,
      text=zoText,
      font=\normalsize,
      rotate=0
    }
  ]
    % Reference envelope for x sin(1/x).
    \addplot[zo reference,domain=-1.2:1.2,samples=2] {x};
    \addplot[zo reference,domain=-1.2:1.2,samples=2] {-x};

    % Secondary graph: g(x)=x sin(1/x), parameterized by u=1/x.
    \addplot[
      zo graph secondary,
      parametric,
      variable=\u,
      domain=\zoUMin:\zoUMax,
      samples=801
    ] ({1/\u},{sin(deg(\u))/\u});
    \addplot[
      zo graph secondary,
      parametric,
      variable=\u,
      domain=\zoUMin:\zoUMax,
      samples=801
    ] ({-1/\u},{sin(deg(\u))/\u});

    % Main graph: f(x)=sin(1/x), split at x=0 and parameterized by u.
    \addplot[
      zo graph main,
      parametric,
      variable=\u,
      domain=\zoUMin:\zoUMax,
      samples=801
    ] ({1/\u},{sin(deg(\u))});
    \addplot[
      zo graph main,
      parametric,
      variable=\u,
      domain=\zoUMin:\zoUMax,
      samples=801
    ] ({-1/\u},{-sin(deg(\u))});

    % Axis labels: every displayed value is an independent node.
    \node[

      yshift=1pt

    enlargelimits=false,
    clip=true,
    clip mode=individual,
    axis lines=middle,

    every axis plot/.append style={line cap=round,line join=round}
  },
  zo graph main/.style={
    draw=zoGraphMain,
    line width=1.2pt,
    solid,
    no marks
  },
  zo reference/.style={
    draw=zoReference,
    line width=0.8pt,
    densely dashed,

  },
  zo point closed/.style={
    only marks,

    draw=zoGraphStrong,

}
